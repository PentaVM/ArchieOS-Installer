cmake_minimum_required(VERSION 3.24)
project(archie-wizard LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.6 REQUIRED COMPONENTS Quick QuickControls2 Network Multimedia)
qt_standard_project_setup(REQUIRES 6.6)
qt_add_executable(archie-wizard main.cpp Wizard.cpp Wizard.h)
qt_add_qml_module(archie-wizard URI Archie.Installer VERSION 1.0 QML_FILES qml/Main.qml)
target_link_libraries(archie-wizard PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Multimedia)
install(TARGETS archie-wizard RUNTIME DESTINATION bin)
